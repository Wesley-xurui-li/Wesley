> {
>   description = "BioNix: GRIDSS/VirusBreakend via Docker (official reproducible workflow)";
> 
>   inputs = {
>     nixpkgs.url = "github:NixOS/nixpkgs/nixos-24.05";
>     flake-utils.url = "github:numtide/flake-utils";
>   };
> 
>   outputs = { self, nixpkgs, flake-utils }:
>     flake-utils.lib.eachDefaultSystem (system:
>     let
>       pkgs = import nixpkgs { inherit system; };
>       runner = pkgs.writeShellScriptBin "virusbreakend-docker" ''
>         set -euo pipefail
>         if ! command -v docker >/dev/null; then
>           echo "docker not found. Please install/start Docker." >&2
>           exit 1
>         fi
>         docker run --rm -it \
>           -v "$PWD:/work" -w /work \
>           gridss/gridss:latest \
>           virusbreakend "$@"
>       '';
>     in
>     {
>       packages.default = runner;
>       apps.default = { type = "app"; program = "${runner}/bin/virusbreakend-docker"; };
>       apps.virusbreakend = { type = "app"; program = "${runner}/bin/virusbreakend-docker"; };
>     });
> }
> EOF
MacBookPro:virusbreakend-nix wesley$ cat > README.md <<'EOF'
> # VirusBreakend (GRIDSS) — Docker-based Nix workflow
> 
> This Nix flake wraps the official [gridss/gridss](https://hub.docker.com/r/gridss/gridss) Docker image for reproducible VirusBreakend and GRIDSS workflows.
> 
> ## Quick Start
> 
> 1. Start Docker Desktop
> 2. Run VirusBreakend help:
>     nix run .#virusbreakend -- --help
> 3. Run with your BAM data:
>     nix run .#virusbreakend -- -o sample.vcf -r ref.fa normal.bam tumor.bam
> 
> ## Notes
> 
> - Files must be in your current directory.
> - Real datasets need ~24GB RAM and ≥4 CPU.
> - Reference FASTA must be BWA-indexed.
> EOF
